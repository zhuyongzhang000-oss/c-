cmake_minimum_required(VERSION 3.20)
project(c_study LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

if(MSVC)
  add_compile_options(/utf-8 /D_CRT_SECURE_NO_WARNINGS)
else()
  add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()

# 每个「环NN_xxx」文件夹自动生成两个启动项（讲义是 讲义.html，用浏览器看）：
#   ringNN_quiz    自测.c（猜输出 + 判断对错，自动判分）
#   ringNN_test    测试.c + 题1~题5（编程题，一次测完）
file(GLOB RING_DIRS LIST_DIRECTORIES true CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/环*")
foreach(dir IN LISTS RING_DIRS)
  if(NOT IS_DIRECTORY "${dir}")
    continue()
  endif()
  get_filename_component(folder "${dir}" NAME)
  string(REGEX MATCH "[0-9][0-9]" nn "${folder}")
  if(nn STREQUAL "")
    continue()
  endif()
  file(GLOB problems CONFIGURE_DEPENDS "${dir}/题*.c")

  foreach(kind quiz test)
    if(kind STREQUAL "quiz")
      set(main_src "${dir}/自测.c")
      set(extra "${dir}/答案/判断答案.c")
    else()
      set(main_src "${dir}/测试.c")
      set(extra ${problems})
    endif()
    if(EXISTS "${main_src}")
      set(target "ring${nn}_${kind}")
      add_executable(${target} "${main_src}" ${extra})
      target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/公共")
      if(NOT MSVC)
        target_link_libraries(${target} PRIVATE m)
      endif()
    endif()
  endforeach()
endforeach()

# 草稿：写完整程序的地方
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/草稿/草稿.c")
  add_executable(draft "${CMAKE_CURRENT_SOURCE_DIR}/草稿/草稿.c")
  target_include_directories(draft PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/公共")
  if(NOT MSVC)
    target_link_libraries(draft PRIVATE m)
  endif()
endif()
