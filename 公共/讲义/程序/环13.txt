struct Stud {
    int num;
    char name[20];
    double score[3];
    double avg;
};
-- main
#define N 5

int main(void)
{
    struct Stud s[N];
    int i;
    for (i = 0; i < N; i++)
        scanf("%d%s%lf%lf%lf", &s[i].num, s[i].name, &s[i].score[0], &s[i].score[1], &s[i].score[2]);
    save_students(s, N, "stud");
    return 0;
}
-- 输入

102 Wang 85 75 95
103 Zhang 90 92 94
104 Liu 60 70 80
105 Chen 88 66 77
