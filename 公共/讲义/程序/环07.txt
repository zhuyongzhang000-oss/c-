        for (j = 0; j < n; j++) scanf("%d", &a[i][j]);

    for (i = 0; i < n; i++) for (j = 0; j < n; j++) scanf("%d", &a[i][j]);
    for (i = 0; i < n; i++) for (j = 0; j < n; j++) scanf("%d", &b[i][j]);
    for (i = 0; i < n; i++) for (j = 0; j < n; j++) scanf("%d", &c[i][j]);
