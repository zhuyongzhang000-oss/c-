-- main
void InputMatrix(int a[][3])
{
    int i, j;
    for (i = 0; i < 3; i++)
        for (j = 0; j < 3; j++) scanf("%d", &a[i][j]);
}

int main(void)
{
    int a[3][3], b[3][3], ta[3][3], tb[3][3], sum[3][3];
    InputMatrix(a);
    InputMatrix(b);
    ReverseMatrix(a, ta);
    ReverseMatrix(b, tb);
    AddMatrix(a, b, sum);
    printf("A 的转置：\n");  OutputMatrix(ta);
    printf("B 的转置：\n");  OutputMatrix(tb);
    printf("A + B：\n");     OutputMatrix(sum);
    return 0;
}
-- 输入
1 2 3
4 5 6
7 8 9
1 1 1
2 2 2
3 3 3
