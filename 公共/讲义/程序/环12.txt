-- 前置
struct Node {
    int num;
    struct Node *next;
};
-- main
/* 建链表（教材例 9.9 的做法）：读到 0 结束 */
struct Node *creat(void)
{
    struct Node *head = NULL, *tail = NULL, *p;
    int num;
    scanf("%d", &num);
    while (num != 0) {
        p = (struct Node *)malloc(sizeof(struct Node));
        p->num = num;
        p->next = NULL;
        if (head == NULL) head = p;
        else tail->next = p;
        tail = p;
        scanf("%d", &num);
    }
    return head;
}

void print_list(struct Node *head)
{
    struct Node *p;
    for (p = head; p != NULL; p = p->next) printf("%d ", p->num);
    printf("\n");
}

int main(void)
{
    struct Node *head;
    int num;
    head = creat();
    scanf("%d", &num);                 /* 要删的学号 */
    head = del(head, num);
    print_list(head);
    return 0;
}
-- 输入
101 102 103 104 0
103
